    function initForm() {

      criteria.forEach(item => {

        const grid = document.getElementById(`grid-${item.cat}`);

        const card = document.createElement('div');

        card.className = 'card';

    function selectRating(id, value) {

      currentRatings[id] = value;

      updateButtonsForField(id);

      calculateDailyScore();

    }

    function updateButtonsForField(id) {

      const container = document.getElementById(`options-${id}`);

      const btns = container.querySelectorAll('.rating-btn');

      btns.forEach((btn, idx) => {

        if (idx + 1 === currentRatings[id]) {

          btn.classList.add('selected');

        } else {

          btn.classList.remove('selected');

        }

      });

    }

    function updateAllButtons() {

      criteria.forEach(item => updateButtonsForField(item.id));

    }

    function calculateDailyScore() {

      let score = 0;

      criteria.forEach(item => {

        const val = currentRatings[item.id] || 3;

        if (item.pos) {

          score += (val - 3) * 5;

        } else {

          score += (3 - val) * 5;

        }

      });

      const preview = document.getElementById('daily-score-preview');

      let statusClass = 'badge-neutral';

      let statusText = 'Equilíbrio / Manutenção';

      if (score >= 10) { statusClass = 'badge-positive'; statusText = 'Superávit Emocional'; }

      else if (score <= -6) { statusClass = 'badge-negative'; statusText = 'Déficit de Energia'; }

      preview.innerHTML = `Saldo do Dia: <span class="score-badge ${statusClass}">${score > 0 ? '+' : ''}${score} pts (${statusText})</span>`;

      return score;

    }

    function saveEntry() {

      const date = document.getElementById('entry-date').value;

      if (!date) return alert('Selecione uma data!');

      const score = calculateDailyScore();

      const db = getStorageData();

      db[date] = { ratings: { ...currentRatings }, score: score };

      localStorage.setItem('mentalmente_db', JSON.stringify(db));

      alert('Registo guardado com sucesso!');

      renderHistory();

      renderCharts();

    }

    function getStorageData() {

      return JSON.parse(localStorage.getItem('mentalmente_db') || '{}');

    }

    function renderHistory() {

      const db = getStorageData();

      const tbody = document.getElementById('history-tbody');

      tbody.innerHTML = '';

      const dates = Object.keys(db).sort().reverse();

      dates.forEach(date => {

        const entry = db[date];

        let statusBadge = '<span class="score-badge badge-neutral">Equilíbrio</span>';

        if (entry.score >= 10) statusBadge = '<span class="score-badge badge-positive">Superávit</span>';

        if (entry.score <= -6) statusBadge = '<span class="score-badge badge-negative">Déficit</span>';

        const tr = document.createElement('tr');

        tr.innerHTML = `

          <td>${date}</td>

          <td><strong>${entry.score > 0 ? '+' : ''}${entry.score} pts</strong></td>

          <td>${statusBadge}</td>

          <td>

            <button class="btn btn-secondary" style="padding:0.3rem 0.6rem; font-size:0.8rem;" onclick="loadEntry('${date}')">✏️ Editar</button>

            <button class="btn btn-danger" style="padding:0.3rem 0.6rem; font-size:0.8rem;" onclick="deleteEntry('${date}')">🗑️ Apagar</button>

          </td>

        `;

        tbody.appendChild(tr);

      });

    }

    function loadEntry(date) {

      const db = getStorageData();

      if (db[date]) {

        currentRatings = { ...db[date].ratings };

        document.getElementById('entry-date').value = date;

        updateAllButtons();

        calculateDailyScore();

        switchTab('registro');

      }

    }

    function deleteEntry(date) {

      if (confirm(`Tem a certeza que deseja apagar o registo de ${date}?`)) {

        const db = getStorageData();

        delete db[date];

        localStorage.setItem('mentalmente_db', JSON.stringify(db));

        renderHistory();

        renderCharts();

      }

    }

    function clearAllData() {

      if (confirm('Atenção: Todos os registos guardados serão permanentemente apagados. Deseja continuar?')) {

        localStorage.removeItem('mentalmente_db');

        renderHistory();

        renderCharts();

      }

    }

    function switchTab(tabId) {

      document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));

      document.querySelectorAll('.tab-content').forEach(content => content.classList.remove('active'));

     

      document.querySelector(`[onclick="switchTab('${tabId}')"]`).classList.add('active');

      document.getElementById(`tab-${tabId}`).classList.add('active');

      if (tabId === 'relatorios') renderCharts();

      if (tabId === 'historico') renderHistory();

    }

    function renderCharts() {

      const db = getStorageData();

      const dates = Object.keys(db).sort();

     

      // Totais Mês

      let totalScore = 0;

      dates.forEach(d => totalScore += db[d].score);

      const avgScore = dates.length ? (totalScore / dates.length).toFixed(1) : 0;

      document.getElementById('monthly-score').innerText = `${totalScore > 0 ? '+' : ''}${totalScore} pts`;

      document.getElementById('monthly-avg').innerText = avgScore;

      // Line Chart

      const lineCtx = document.getElementById('lineChart').getContext('2d');

      if (lineChart) lineChart.destroy();

      lineChart = new Chart(lineCtx, {

        type: 'line',

        data: {

          labels: dates,

          datasets: [{

            label: 'Saldo Emocional Diário',

            data: dates.map(d => db[d].score),

            borderColor: '#6366f1',

            backgroundColor: 'rgba(99, 102, 241, 0.2)',

            fill: true,

            tension: 0.3

          }]

        },

        options: { responsive: true, maintainAspectRatio: false }

      });

      // Radar Chart

      const avgRatings = criteria.map(c => {

        if (!dates.length) return 3;

        const sum = dates.reduce((acc, d) => acc + (db[d].ratings[c.id] || 3), 0);

        return (sum / dates.length).toFixed(2);

      });

      const radarCtx = document.getElementById('radarChart').getContext('2d');

      if (radarChart) radarChart.destroy();

      radarChart = new Chart(radarCtx, {

        type: 'radar',

        data: {

          labels: criteria.map(c => c.title),

          datasets: [{

            label: 'Média das Dimensões',

            data: avgRatings,

            borderColor: '#22c55e',

            backgroundColor: 'rgba(34, 197, 94, 0.2)'

          }]

        },

        options: {

          responsive: true,

          maintainAspectRatio: false,

          scales: { r: { min: 1, max: 5 } }

        }

      });

    }

    function generateTestData() {

      const db = {};

      const today = new Date();

      for (let i = 29; i >= 0; i--) {

        const d = new Date(today);

        d.setDate(d.getDate() - i);

        const dateStr = d.toISOString().split('T')[0];

       

        const ratings = {};

        let score = 0;

        criteria.forEach(c => {

          const val = Math.floor(Math.random() * 5) + 1;

          ratings[c.id] = val;

          if (c.pos) score += (val - 3) * 5;

          else score += (3 - val) * 5;

        });

        db[dateStr] = { ratings, score };

      }

      localStorage.setItem('mentalmente_db', JSON.stringify(db));

      alert('30 dias de dados de teste gerados com sucesso!');

      renderHistory();

    }

    function exportJSON() {

      const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(getStorageData(), null, 2));

      const downloadAnchor = document.createElement('a');

      downloadAnchor.setAttribute("href", dataStr);

      downloadAnchor.setAttribute("download", "mentalmente_backup.json");

      document.body.appendChild(downloadAnchor);

      downloadAnchor.click();

      downloadAnchor.remove();

    }
